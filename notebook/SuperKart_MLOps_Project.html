"""Step 1 - Data registration.

Registers the raw SuperKart dataset: validates that every expected column is
present (with a sensible dtype), prints a short summary, and writes a small
registry record (data/registry.json) so later steps can see which dataset
version they are working on.

Usage (from the repo root):
    python src/register_data.py
"""
import hashlib
import json
import sys
from pathlib import Path

import pandas as pd

DATA_PATH = Path("data/SuperKart.csv")
REGISTRY_PATH = Path("data/registry.json")

EXPECTED_COLUMNS = {
    "Product_Id": "object",
    "Product_Weight": "number",
    "Product_Sugar_Content": "object",
    "Product_Allocated_Area": "number",
    "Product_Type": "object",
    "Product_MRP": "number",
    "Store_Id": "object",
    "Store_Establishment_Year": "number",
    "Store_Size": "object",
    "Store_Location_City_Type": "object",
    "Store_Type": "object",
    "Product_Store_Sales_Total": "number",
}


def validate(df: pd.DataFrame) -> list:
    """Return a list of problems found (empty list = dataset is valid)."""
    problems = []

    missing = [c for c in EXPECTED_COLUMNS if c not in df.columns]
    extra = [c for c in df.columns if c not in EXPECTED_COLUMNS]
    if missing:
        problems.append(f"Missing columns: {missing}")
    if extra:
        problems.append(f"Unexpected columns: {extra}")

    for col, kind in EXPECTED_COLUMNS.items():
        if col not in df.columns:
            continue
        is_numeric = pd.api.types.is_numeric_dtype(df[col])
        if kind == "number" and not is_numeric:
            problems.append(f"Column {col} should be numeric, got {df[col].dtype}")
        if kind == "object" and is_numeric:
            problems.append(f"Column {col} should be text, got {df[col].dtype}")

    if len(df) == 0:
        problems.append("Dataset has no rows")
    return problems


def main() -> int:
    if not DATA_PATH.exists():
        print(f"ERROR: dataset not found at {DATA_PATH}")
        return 1

    df = pd.read_csv(DATA_PATH)
    problems = validate(df)

    print("=" * 60)
    print("DATASET REGISTRATION - SuperKart")
    print("=" * 60)
    print(f"File           : {DATA_PATH}")
    print(f"Rows x columns : {df.shape[0]} x {df.shape[1]}")
    print(f"Missing values : {int(df.isna().sum().sum())}")
    print(f"Duplicate rows : {int(df.duplicated().sum())}")
    print("\nColumn dtypes:")
    print(df.dtypes.to_string())
    print("\nTarget (Product_Store_Sales_Total) summary:")
    print(df["Product_Store_Sales_Total"].describe().round(2).to_string())

    if problems:
        print("\nVALIDATION FAILED:")
        for p in problems:
            print(f"  - {p}")
        return 1

    record = {
        "dataset": "SuperKart",
        "path": str(DATA_PATH),
        "rows": int(df.shape[0]),
        "columns": list(df.columns),
        "sha256": hashlib.sha256(DATA_PATH.read_bytes()).hexdigest(),
    }
    REGISTRY_PATH.write_text(json.dumps(record, indent=2))
    print(f"\nVALIDATION PASSED - all {len(EXPECTED_COLUMNS)} expected columns present.")
    print(f"Registered dataset (sha256 {record['sha256'][:12]}...) -> {REGISTRY_PATH}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

4. Data preparation

src/data_prep.py loads the CSV directly from the repository data folder and then:

  1. removes duplicates (none found),
  2. merges reg into Regular,
  3. derives Product_Family from the first two letters of Product_Id,
  4. drops Product_Id and Store_Id (identifier columns with no extra information),
  5. splits 80/20 into train and test with a fixed seed and saves both to data/processed/.

In GitHub Actions, data/processed/ is uploaded as the processed-data artifact and downloaded by the training job.

In [10]:
print(Path('src/data_prep.py').read_text())
"""Step 2 - Data preparation.

Loads the raw CSV straight from the repository data folder, cleans it, drops
columns that should not be model inputs, splits into train/test and saves the
splits to data/processed/. The processed folder is what the GitHub Actions
workflow uploads as an artifact for the training job.

Usage (from the repo root):
    python src/data_prep.py
"""
from pathlib import Path

import pandas as pd
from sklearn.model_selection import train_test_split

RAW_PATH = Path("data/SuperKart.csv")
OUT_DIR = Path("data/processed")
TARGET = "Product_Store_Sales_Total"
RANDOM_STATE = 42
TEST_SIZE = 0.2


def clean(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()

    # 1. Remove exact duplicates (none expected, but keeps the step safe to re-run)
    df = df.drop_duplicates()

    # 2. Fix inconsistent labels: 'reg' and 'Regular' are the same sugar category
    df["Product_Sugar_Content"] = (
        df["Product_Sugar_Content"].str.strip().replace({"reg": "Regular"})
    )

    # 3. Product_Id starts with two letters that mark the broad product family
    #    (FD = food, DR = drink, NC = non-consumable). Keep that, drop the id.
    df["Product_Family"] = df["Product_Id"].str[:2]

    # 4. Drop unnecessary columns:
    #    - Product_Id : unique per row, no predictive value once family is extracted
    #    - Store_Id   : fully determined by Store_Size / City_Type / Store_Type /
    #                   Establishment_Year, so it only duplicates information
    df = df.drop(columns=["Product_Id", "Store_Id"])

    # 5. Drop rows with missing values (the dataset has none, guard for new data)
    df = df.dropna()
    return df


def main() -> None:
    df = pd.read_csv(RAW_PATH)
    print(f"Loaded raw data: {df.shape}")

    df = clean(df)
    print(f"Cleaned data   : {df.shape}")
    print("Sugar labels   :", sorted(df["Product_Sugar_Content"].unique()))

    train, test = train_test_split(df, test_size=TEST_SIZE, random_state=RANDOM_STATE)

    OUT_DIR.mkdir(parents=True, exist_ok=True)
    train.to_csv(OUT_DIR / "train.csv", index=False)
    test.to_csv(OUT_DIR / "test.csv", index=False)
    print(f"Saved {OUT_DIR / 'train.csv'} {train.shape} and {OUT_DIR / 'test.csv'} {test.shape}")


if __name__ == "__main__":
    main()

5. Model building with experiment tracking

src/train.py loads the saved train/test splits, wraps one-hot encoding and the model in a single scikit-learn Pipeline (so the app can pass raw columns), and tunes two candidates with 3-fold GridSearchCV scored on R2:

ModelParameters tuned
Random Forestn_estimators (100, 200), max_depth (6, 10, 14), min_samples_leaf (1, 3, 5)
Gradient Boostingn_estimators (100, 200), learning_rate (0.05, 0.1), max_depth (2, 3, 4)

Tracking: every one of the 30 parameter combinations is logged to experiments/tuning_trials.csv with its cross-validated R2, and the best parameters and train/test metrics per model go to experiments/run_summary.json. MLflow logging is used as well when the package is installed (it is listed in requirements.txt for the CI runner). The best model is saved to model/best_model.joblib.

In [13]:
print(Path('src/train.py').read_text())
"""Step 3 - Model building with experiment tracking.

Loads the train/test splits (downloaded from the workflow artifact in CI, or
produced locally by data_prep.py), tunes two tree-ensemble models with
GridSearchCV, logs every parameter combination tried, evaluates on the held-out
test set and saves the best model to model/best_model.joblib so the pipeline
can commit it to the repository.

Tracking: every run is always written to experiments/ (CSV + JSON) so the log
is committed with the repo. If MLflow is installed it is used as well.

Usage (from the repo root):
    python src/train.py
"""
import json
from pathlib import Path

import joblib
import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.ensemble import GradientBoostingRegressor, RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from sklearn.model_selection import GridSearchCV
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder

try:  # optional - not required for the pipeline to work
    import mlflow
except ImportError:  # pragma: no cover
    mlflow = None

DATA_DIR = Path("data/processed")
MODEL_DIR = Path("model")
EXP_DIR = Path("experiments")
TARGET = "Product_Store_Sales_Total"
RANDOM_STATE = 42

NUMERIC = ["Product_Weight", "Product_Allocated_Area", "Product_MRP", "Store_Establishment_Year"]
CATEGORICAL = [
    "Product_Sugar_Content",
    "Product_Type",
    "Store_Size",
    "Store_Location_City_Type",
    "Store_Type",
    "Product_Family",
]

# Models and the parameter grids to tune
CANDIDATES = {
    "random_forest": (
        RandomForestRegressor(random_state=RANDOM_STATE, n_jobs=-1),
        {
            "model__n_estimators": [100, 200],
            "model__max_depth": [6, 10, 14],
            "model__min_samples_leaf": [1, 3, 5],
        },
    ),
    "gradient_boosting": (
        GradientBoostingRegressor(random_state=RANDOM_STATE),
        {
            "model__n_estimators": [100, 200],
            "model__learning_rate": [0.05, 0.1],
            "model__max_depth": [2, 3, 4],
        },
    ),
}


def build_pipeline(model) -> Pipeline:
    """Preprocessing + model in one object, so the app can feed raw columns."""
    pre = ColumnTransformer(
        [
            ("cat", OneHotEncoder(handle_unknown="ignore"), CATEGORICAL),
            ("num", "passthrough", NUMERIC),
        ]
    )
    return Pipeline([("pre", pre), ("model", model)])


def evaluate(model, X, y) -> dict:
    pred = model.predict(X)
    return {
        "r2": float(r2_score(y, pred)),
        "rmse": float(np.sqrt(mean_squared_error(y, pred))),
        "mae": float(mean_absolute_error(y, pred)),
    }


def main() -> None:
    train = pd.read_csv(DATA_DIR / "train.csv")
    test = pd.read_csv(DATA_DIR / "test.csv")
    X_train, y_train = train.drop(columns=[TARGET]), train[TARGET]
    X_test, y_test = test.drop(columns=[TARGET]), test[TARGET]
    print(f"Train {X_train.shape}, Test {X_test.shape}")

    EXP_DIR.mkdir(exist_ok=True)
    MODEL_DIR.mkdir(exist_ok=True)
    if mlflow is not None:
        mlflow.set_tracking_uri(f"file:{EXP_DIR.resolve() / 'mlruns'}")
        mlflow.set_experiment("superkart-sales-forecast")

    all_trials = []
    summary = {}
    best_name, best_search, best_r2 = None, None, -np.inf

    for name, (model, grid) in CANDIDATES.items():
        print(f"\nTuning {name} ...")
        search = GridSearchCV(
            build_pipeline(model), grid, cv=3, scoring="r2", n_jobs=-1, refit=True
        )
        search.fit(X_train, y_train)

        # Log every parameter combination that was tried
        cv = pd.DataFrame(search.cv_results_)
        for _, row in cv.iterrows():
            trial = {"model": name, **{k.replace("model__", ""): v for k, v in row["params"].items()}}
            trial["cv_r2_mean"] = float(row["mean_test_score"])
            trial["cv_r2_std"] = float(row["std_test_score"])
            all_trials.append(trial)

        test_metrics = evaluate(search.best_estimator_, X_test, y_test)
        train_metrics = evaluate(search.best_estimator_, X_train, y_train)
        best_params = {k.replace("model__", ""): v for k, v in search.best_params_.items()}
        summary[name] = {
            "best_params": best_params,
            "cv_r2": float(search.best_score_),
            "train": train_metrics,
            "test": test_metrics,
        }
        print(f"  best params : {best_params}")
        print(f"  CV R2       : {search.best_score_:.4f}")
        print(f"  Test        : {test_metrics}")

        if mlflow is not None:
            with mlflow.start_run(run_name=name):
                mlflow.log_params(best_params)
                mlflow.log_metric("cv_r2", float(search.best_score_))
                for split, m in (("train", train_metrics), ("test", test_metrics)):
                    for k, v in m.items():
                        mlflow.log_metric(f"{split}_{k}", v)

        if test_metrics["r2"] > best_r2:
            best_name, best_search, best_r2 = name, search, test_metrics["r2"]

    # Save logs, metrics and the best model
    pd.DataFrame(all_trials).to_csv(EXP_DIR / "tuning_trials.csv", index=False)
    (EXP_DIR / "run_summary.json").write_text(json.dumps(summary, indent=2))

    # compress=3 keeps the file well under GitHub's 25 MB web-upload limit
    joblib.dump(best_search.best_estimator_, MODEL_DIR / "best_model.joblib", compress=3)
    final = {"best_model": best_name, **summary[best_name]}
    (MODEL_DIR / "metrics.json").write_text(json.dumps(final, indent=2))

    print("\n" + "=" * 60)
    print(f"BEST MODEL: {best_name}")
    print(f"Test R2={final['test']['r2']:.4f}  RMSE={final['test']['rmse']:.2f}  MAE={final['test']['mae']:.2f}")
    print(f"Saved model -> {MODEL_DIR / 'best_model.joblib'}")
    print(f"Saved tuning log ({len(all_trials)} trials) -> {EXP_DIR / 'tuning_trials.csv'}")


if __name__ == "__main__":
    main()

Observations

  • Both ensembles fit well. The tuned Random Forest is the best model (test R2 about 0.932, RMSE about 278, MAE about 106) and edges out Gradient Boosting (test R2 about 0.926).
  • Cross-validated R2 (about 0.928) is close to test R2, so there is no sign of overfitting to the test split. The train R2 is higher, as expected for a deep forest.
  • Product MRP is by far the most important feature (about 45% of the importance), then product weight, followed by the store descriptors (Food Mart, Tier 3, Small), which agrees with the EDA.
  • Average absolute error is highest for the top quarter of sales values (about 176) and the bottom quarter (about 116), and lowest in the middle (about 59 to 74), so the model is least reliable at the extremes.

7. Automated GitHub Actions workflow

.github/workflows/pipeline.yml runs the whole pipeline on every push to main (and on demand). The train job commits the model and tuning logs back to main; the paths-ignore filter and the [skip ci] commit message stop that commit from triggering the workflow again.

9. Summary and business insights

  • Result: a tuned Random Forest explains about 93% of the variance in product-store sales (test R2 0.932, RMSE about 278, MAE about 106 on a mean of about 3,464).
  • What drives sales: product price (MRP) and weight dominate, and the store itself matters a lot. Departmental Stores and Tier 1 locations sell far more per product than Food Marts in Tier 3 cities.
  • Planning use: the supply chain team can use predicted sales per product and store to size procurement, and a Food Mart in a Tier 3 city should be planned with roughly a third of the volume of a Departmental Store for the same product.
  • Data quality: the reg / Regular label clash was fixed in the pipeline, so any new data with the same issue is cleaned automatically.
  • Limits: the dataset has only four stores and no dates, so the model predicts total sales for a product-store combination and cannot produce weekly, monthly or quarterly forecasts or capture trends over time. Predictions for a new store are extrapolations from four store profiles. Getting dated sales history would be the most valuable next step.
  • MLOps: every push re-validates the data, rebuilds the splits, retrains and tunes, commits the best model and keeps the app in sync, with no manual steps.
